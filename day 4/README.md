SQL-запросы для Варианта 3 (Библиотека)
1. Простые выборки
sql
-- 1. Вывести все книги
SELECT * FROM Книги;

-- 2. Вывести книги с количеством экземпляров > 3
SELECT * FROM Книги WHERE количество_экземпляров > 3;

-- 3. Вывести читателей, отсортированных по фамилии
SELECT * FROM Читатели ORDER BY фамилия;

-- 4. Вывести книги, название которых начинается с 'В'
SELECT * FROM Книги WHERE название LIKE 'В%';

-- 5. Вывести первые 5 записей о выдаче
SELECT * FROM Выдача_книг LIMIT 5;
2. Запросы с JOIN
sql
-- 6. Выдать информацию о выдачах с именами читателей и названиями книг
SELECT в.дата_выдачи, ч.фамилия, ч.имя, к.название
FROM Выдача_книг в
JOIN Читатели ч ON в.id_читателя = ч.id_читателя
JOIN Книги к ON в.id_книги = к.id_книги;

-- 7. Выдать авторов книг с их книгами
SELECT а.фамилия AS автор, к.название AS книга
FROM Книги_Авторы ка
JOIN Авторы а ON ка.id_автора = а.id_автора
JOIN Книги к ON ка.id_книги = к.id_книги;

-- 8. Выдать читателей, у которых нет штрафов
SELECT ч.*
FROM Читатели ч
LEFT JOIN Штрафы ш ON ч.id_читателя = ш.id_читателя
WHERE ш.id_штрафа IS NULL;

-- 9. Подсчитать количество книг у каждого автора
SELECT а.фамилия, COUNT(ка.id_книги) AS количество_книг
FROM Авторы а
LEFT JOIN Книги_Авторы ка ON а.id_автора = ка.id_автора
GROUP BY а.id_автора;

-- 10. Найти читателей с просроченными книгами
SELECT DISTINCT ч.фамилия, ч.имя
FROM Выдача_книг в
JOIN Читатели ч ON в.id_читателя = ч.id_читателя
WHERE в.дата_возврата_факт IS NULL AND в.дата_возврата_план < CURDATE();
3. Агрегация и GROUP BY
sql
-- 11. Общее количество книг в библиотеке
SELECT SUM(количество_экземпляров) AS всего_книг FROM Книги;

-- 12. Статистика по книгам
SELECT 
    COUNT(*) AS количество_книг,
    AVG(количество_экземпляров) AS среднее_экземпляров,
    MAX(год) AS последний_год,
    MIN(год) AS первый_год
FROM Книги;

-- 13. Количество выдач по месяцам
SELECT DATE_FORMAT(дата_выдачи, '%Y-%m') AS месяц, COUNT(*) AS выдано
FROM Выдача_книг
GROUP BY DATE_FORMAT(дата_выдачи, '%Y-%m');

-- 14. Авторы с количеством книг больше 1
SELECT а.фамилия, COUNT(ка.id_книги) AS количество_книг
FROM Авторы а
JOIN Книги_Авторы ка ON а.id_автора = ка.id_автора
GROUP BY а.id_автора
HAVING COUNT(ка.id_книги) > 1;
4. Подзапросы
sql
-- 15. Книги, изданные после года издания самой старой книги
SELECT * FROM Книги 
WHERE год > (SELECT MIN(год) FROM Книги);

-- 16. Читатели, у которых нет штрафов
SELECT * FROM Читатели 
WHERE id_читателя NOT IN (SELECT id_читателя FROM Штрафы);

-- 17. Читатели, у которых есть выданные книги
SELECT * FROM Читатели 
WHERE EXISTS (SELECT 1 FROM Выдача_книг WHERE id_читателя = Читатели.id_читателя AND дата_возврата_факт IS NULL);
5. Обновление и удаление
sql
-- 18. Увеличить количество экземпляров всех книг на 1
UPDATE Книги SET количество_экземпляров = количество_экземпляров + 1;

-- 19. Обновить статус выдачи при возврате
UPDATE Выдача_книг 
SET дата_возврата_факт = CURDATE(), статус = 'Возвращена'
WHERE id_выдачи = 1;

-- 20. Удалить старые штрафы
DELETE FROM Штрафы WHERE оплачен = TRUE;
