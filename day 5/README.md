1-5. Создание БД и таблиц

sql
-- Создание базы данных
CREATE DATABASE IF NOT EXISTS Variant3_Work;
USE Variant3_Work;

-- Таблица Книги
CREATE TABLE Книги (
    id_книги INT PRIMARY KEY AUTO_INCREMENT,
    название VARCHAR(200) NOT NULL,
    год INT,
    isbn VARCHAR(20) UNIQUE,
    количество_экземпляров INT NOT NULL DEFAULT 1
);

-- Таблица Авторы
CREATE TABLE Авторы (
    id_автора INT PRIMARY KEY AUTO_INCREMENT,
    имя VARCHAR(100) NOT NULL,
    фамилия VARCHAR(100) NOT NULL
);

-- Таблица Книги_Авторы (связь M:N)
CREATE TABLE Книги_Авторы (
    id_книги INT,
    id_автора INT,
    PRIMARY KEY (id_книги, id_автора),
    FOREIGN KEY (id_книги) REFERENCES Книги(id_книги),
    FOREIGN KEY (id_автора) REFERENCES Авторы(id_автора)
);

-- Таблица Читатели
CREATE TABLE Читатели (
    id_читателя INT PRIMARY KEY AUTO_INCREMENT,
    фамилия VARCHAR(50) NOT NULL,
    имя VARCHAR(50) NOT NULL,
    отчество VARCHAR(50),
    телефон VARCHAR(20),
    адрес VARCHAR(200),
    дата_регистрации DATE
);

-- Таблица Выдача_книг
CREATE TABLE Выдача_книг (
    id_выдачи INT PRIMARY KEY AUTO_INCREMENT,
    id_книги INT,
    id_читателя INT,
    дата_выдачи DATE NOT NULL,
    дата_возврата_план DATE NOT NULL,
    дата_возврата_факт DATE,
    статус VARCHAR(20),
    FOREIGN KEY (id_книги) REFERENCES Книги(id_книги),
    FOREIGN KEY (id_читателя) REFERENCES Читатели(id_читателя)
);

-- Таблица Штрафы
CREATE TABLE Штрафы (
    id_штрафа INT PRIMARY KEY AUTO_INCREMENT,
    id_читателя INT,
    причина VARCHAR(200),
    сумма DECIMAL(8,2),
    оплачен BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (id_читателя) REFERENCES Читатели(id_читателя)
);

12-25. Запросы и операции

sql
-- 12. Вывести книги, изданные после 2000 года
SELECT * FROM Книги WHERE год > 2000;

-- 13. Вывести список читателей, отсортированных по фамилии
SELECT * FROM Читатели ORDER BY фамилия;

-- 14. Найти книги, в названии которых есть слово «мир»
SELECT * FROM Книги WHERE название LIKE '%мир%';

-- 15. Вывести все выдачи с указанием фамилии читателя и названия книги
SELECT в.дата_выдачи, ч.фамилия, ч.имя, к.название
FROM Выдача_книг в
JOIN Читатели ч ON в.id_читателя = ч.id_читателя
JOIN Книги к ON в.id_книги = к.id_книги;

-- 16. Для каждой книги посчитать, сколько раз её брали
SELECT к.название, COUNT(в.id_выдачи) AS количество_выдач
FROM Книги к
LEFT JOIN Выдача_книг в ON к.id_книги = в.id_книги
GROUP BY к.id_книги
HAVING COUNT(в.id_выдачи) > 0;

-- 17. Найти читателя, который брал книги чаще всего
SELECT ч.фамилия, ч.имя, COUNT(в.id_выдачи) AS количество_выдач
FROM Читатели ч
JOIN Выдача_книг в ON ч.id_читателя = в.id_читателя
GROUP BY ч.id_читателя
ORDER BY количество_выдач DESC
LIMIT 1;

-- 18. Вывести среднюю дату возврата по всем выдачам, где возврат был
SELECT AVG(DATEDIFF(дата_возврата_факт, дата_выдачи)) AS средняя_длительность
FROM Выдача_книг
WHERE дата_возврата_факт IS NOT NULL;

-- 19. Сгруппировать выдачи по месяцам и подсчитать количество
SELECT DATE_FORMAT(дата_выдачи, '%Y-%m') AS месяц, COUNT(*) AS количество
FROM Выдача_книг
GROUP BY DATE_FORMAT(дата_выдачи, '%Y-%m');

-- 20. Найти книги, которые ни разу не выдавались
SELECT * FROM Книги
WHERE id_книги NOT IN (SELECT DISTINCT id_книги FROM Выдача_книг);

-- 21. Установить штраф в размере 50 руб. для всех неоплаченных штрафов
UPDATE Штрафы SET сумма = 50 WHERE оплачен = FALSE;

-- 22. Удалить книгу с id=5 (если у неё нет выдач)
DELETE FROM Книги_Авторы WHERE id_книги = 5;
DELETE FROM Книги WHERE id_книги = 5;

-- 23. Добавить в таблицу Книги поле pages (INT)
ALTER TABLE Книги ADD COLUMN pages INT;

-- 24. Создать представление OverdueLoans
CREATE VIEW OverdueLoans AS
SELECT ч.фамилия, ч.имя, к.название, в.дата_выдачи, в.дата_возврата_план
FROM Выдача_книг в
JOIN Читатели ч ON в.id_читателя = ч.id_читателя
JOIN Книги к ON в.id_книги = к.id_книги
WHERE в.дата_возврата_факт IS NULL AND в.дата_возврата_план < CURDATE();

-- 25. Сложный запрос: для каждого автора вывести количество книг,
-- общее количество выдач этих книг и среднюю длительность чтения
SELECT 
    а.фамилия AS автор,
    COUNT(DISTINCT к.id_книги) AS количество_книг,
    COUNT(в.id_выдачи) AS всего_выдач,
    AVG(DATEDIFF(в.дата_возврата_факт, в.дата_выдачи)) AS средняя_длительность_чтения
FROM Авторы а
LEFT JOIN Книги_Авторы ка ON а.id_автора = ка.id_автора
LEFT JOIN Книги к ON ка.id_книги = к.id_книги
LEFT JOIN Выдача_книг в ON к.id_книги = в.id_книги
WHERE в.дата_возврата_факт IS NOT NULL
GROUP BY а.id_автора;
